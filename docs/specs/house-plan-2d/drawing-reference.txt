# ---------------------------------------------------------------- drawing helpers
INK = "#3b2e24"


def jit(rng, a=0.9):
    return rng.uniform(-a, a)


def sketch_line(rng, x1, y1, x2, y2, w=1.3, extra=""):
    """Two slightly different strokes with overshoot: the hand-drawn ink line of the sketch."""
    out = []
    for k in range(2):
        dx, dy = x2 - x1, y2 - y1
        n = max((dx * dx + dy * dy) ** 0.5, 1)
        ox, oy = dx / n * 3, dy / n * 3  # overshoot at both ends
        mx, my = (x1 + x2) / 2 + jit(rng, 1.2), (y1 + y2) / 2 + jit(rng, 1.2)
        out.append(
            f'<path d="M{x1 - ox + jit(rng):.1f} {y1 - oy + jit(rng):.1f} Q{mx:.1f} {my:.1f} '
            f'{x2 + ox + jit(rng):.1f} {y2 + oy + jit(rng):.1f}" stroke-width="{w if k == 0 else w * 0.6:.2f}" '
            f'opacity="{0.95 if k == 0 else 0.55}"{extra}/>'
        )
    return "".join(out)


def sketch_rect(rng, x, y, w, h, sw=1.3):
    return (sketch_line(rng, x, y, x + w, y, sw) + sketch_line(rng, x + w, y, x + w, y + h, sw)
            + sketch_line(rng, x + w, y + h, x, y + h, sw) + sketch_line(rng, x, y + h, x, y, sw))


def tree(rng, cx, cy, r, cls="plan-tree"):
    spokes = "".join(
        f'<path d="M{cx:.1f} {cy:.1f} L{cx + r * 0.92 * c:.1f} {cy + r * 0.92 * s:.1f}"/>'
        for c, s in [(__import__("math").cos(a), __import__("math").sin(a))
                     for a in [k * 6.283 / 9 + rng.uniform(0, 0.4) for k in range(9)]]
    )
    return (f'<g class="{cls}"><circle cx="{cx:.1f}" cy="{cy:.1f}" r="{r:.1f}" class="plan-tree__wash" filter="url(#plan-wash)"/>'
            f'<circle cx="{cx:.1f}" cy="{cy:.1f}" r="{r:.1f}" class="plan-tree__ring"/>'
            f'<g class="plan-tree__spokes">{spokes}</g></g>')


def plan_svg(lang, orient):
    """The plan, horizontal (front at left) or vertical (front at top)."""
    t = T[lang]
    rng = random.Random(7)
    pad = 26
    if orient == "h":
        s, su = 3.3, 3.3
        def P(u, v):
            return pad + 34 + u * su, pad + v * s
        def R(u0, u1, v0, v1):
            x0, y0 = P(u0, v0); x1, y1 = P(u1, v1)
            return x0, y0, x1 - x0, y1 - y0
        vw, vh = pad * 2 + 34 + L * su, pad * 2 + W * s + 14
    else:
        s, su = 4.3, 3.0
        def P(u, v):
            return pad + v * s, pad + 34 + u * su
        def R(u0, u1, v0, v1):
            x0, y0 = P(u0, v0); x1, y1 = P(u1, v1)
            return x0, y0, x1 - x0, y1 - y0
        vw, vh = pad * 2 + W * s, pad * 2 + 34 + L * su + 14

    g_wash, g_ink, g_hit, g_text = [], [], [], []
    # trees around the plot (outside), like the sketch
    for (u, v, r) in [(-4, -6, 15), (30, 88, 13), (120, -9, 12), (232, 90, 16), (256, 14, 14), (176, 89, 11)]:
        x, y = P(u, v)
        g_wash.append(tree(rng, x, y, r * (1 if orient == "h" else 1.1), "plan-tree plan-tree--out"))
    for key, kind, (u0, u1, v0, v1), photo, en, vi, _, _ in ROOMS:
        x, y, w, h = R(u0, u1, v0, v1)
        name = en if lang == "en" else vi
        if kind == "garden":
            g_wash.append(f'<rect x="{x + 3:.1f}" y="{y + 3:.1f}" width="{w - 6:.1f}" height="{h - 6:.1f}" class="plan-wash plan-wash--garden" filter="url(#plan-wash)"/>')
            # planting in the garden
            for k in range(3 if key == "front-garden" else 4):
                tx = x + w * rng.uniform(0.2, 0.8); ty = y + h * (0.2 + 0.6 * k / 3 if orient == "h" else rng.uniform(0.25, 0.75))
                if orient == "v":
                    tx = x + w * (0.15 + 0.7 * k / 3); ty = y + h * rng.uniform(0.3, 0.7)
                g_wash.append(tree(rng, tx, ty, rng.uniform(8, 12)))
            g_ink.append(f'<g class="plan-wall plan-wall--fence">{sketch_rect(rng, x, y, w, h, 1.1)}</g>')
        else:
            cls = {"room": "plan-wash--room", "wet": "plan-wash--wet", "loft": "plan-wash--loft"}[kind]
            if kind != "loft":
                g_wash.append(f'<rect x="{x + 2:.1f}" y="{y + 2:.1f}" width="{w - 4:.1f}" height="{h - 4:.1f}" class="plan-wash {cls}" filter="url(#plan-wash)"/>'
                              f'<rect x="{x + 4:.1f}" y="{y + 4:.1f}" width="{w - 8:.1f}" height="{h - 8:.1f}" class="plan-edge {cls}" filter="url(#plan-wash)"/>')
            if kind == "wet":
                g_wash.append(f'<rect x="{x:.1f}" y="{y:.1f}" width="{w:.1f}" height="{h:.1f}" fill="url(#plan-hatch)"/>')
            if kind == "loft":
                g_ink.append(f'<rect x="{x:.1f}" y="{y:.1f}" width="{w:.1f}" height="{h:.1f}" class="plan-loft"/>')
            else:
                g_ink.append(f'<g class="plan-wall">{sketch_rect(rng, x, y, w, h, 2.0)}</g>')
        # the tap target and its label
        g_hit.append(
            f'<a class="plan__room" href="#room-{key}" data-room="{key}" aria-label="{name}">'
            f'<rect class="plan__hit" x="{x:.1f}" y="{y:.1f}" width="{w:.1f}" height="{h:.1f}"/>'
        )
        cx, cy = x + w / 2, y + h / 2
        if key == "living-kitchen":
            cx, cy = P(137, 40)
        if key == "laundry" and orient == "h":
            label = f'<text class="plan__label" x="{cx:.1f}" y="{cy:.1f}" transform="rotate(-90 {cx:.1f} {cy:.1f})">{name}</text>'
        else:
            words = name.split(" ")
            lines, cur = [], ""
            maxc = 12 if key == "living-kitchen" else max(int(w / 7.2), 6)
            for wd in words:
                if len(cur) + len(wd) + 1 > maxc and cur:
                    lines.append(cur); cur = wd
                else:
                    cur = (cur + " " + wd).strip()
            lines.append(cur)
            y0 = cy - (len(lines) - 1) * 8
            if key == "light-well":
                y0 = y + h - 10 - (len(lines) - 1) * 16 if orient == "h" else y + h - 8 - (len(lines) - 1) * 16
            label = "".join(f'<text class="plan__label" x="{cx:.1f}" y="{y0 + n * 16:.1f}">{ln}</text>' for n, ln in enumerate(lines))
        g_hit.append(label + "</a>")

    # details: light well round wall + opening, skylights, kitchen counter and table, beds, door gaps
    x, y, w, h = R(50, 90, 0, 34)
    r = min(w, h) * 0.34
    lcx, lcy = x + w / 2, y + h / 2 - (6 if orient == "h" else 4)
    g_ink.append(f'<circle cx="{lcx:.1f}" cy="{lcy:.1f}" r="{r:.1f}" class="plan-round" filter="url(#plan-wash)"/>'
                 f'<circle cx="{lcx:.1f}" cy="{lcy:.1f}" r="{r:.1f}" class="plan-round__ink"/>'
                 f'<circle cx="{lcx:.1f}" cy="{lcy:.1f}" r="2.2" fill="{INK}"/>')
    for (u0, u1, v0, v1) in [(130, 142, 12, 26), (130, 142, 54, 68)]:
        sx, sy, sw_, sh = R(u0, u1, v0, v1)
        g_ink.append(f'<rect x="{sx:.1f}" y="{sy:.1f}" width="{sw_:.1f}" height="{sh:.1f}" class="plan-sky"/>')
    kx, ky, kw, kh = R(140, 153, 50, 78)  # kitchen counter along the wall
    g_ink.append(f'<rect x="{kx:.1f}" y="{ky:.1f}" width="{kw:.1f}" height="{kh:.1f}" class="plan-furn"/>')
    tx, ty = P(116, 64)
    g_ink.append(f'<circle cx="{tx:.1f}" cy="{ty:.1f}" r="13" class="plan-furn"/>')
    for (u0, u1, v0, v1) in [(178, 198, 6, 26), (178, 198, 56, 76)]:
        bx, by, bw, bh = R(u0, u1, v0, v1)
        g_ink.append(f'<rect x="{bx:.1f}" y="{by:.1f}" width="{bw:.1f}" height="{bh:.1f}" rx="2" class="plan-furn"/>')
    # stair to the loft
    for k in range(6):
        sx0, sy0 = P(96 + k * 3, 42); sx1, sy1 = P(96 + k * 3, 50)
        g_ink.append(f'<path d="M{sx0:.1f} {sy0:.1f} L{sx1:.1f} {sy1:.1f}" class="plan-stair"/>')
    # the lane and the arrow in, like the sketch
    ax, ay = P(-12, 40)
    if orient == "h":
        arrow = f'<path d="M{ax - 6:.1f} {ay:.1f} L{ax + 16:.1f} {ay:.1f} M{ax + 9:.1f} {ay - 6:.1f} L{ax + 16:.1f} {ay:.1f} L{ax + 9:.1f} {ay + 6:.1f}" class="plan-arrow"/>'
        lane = f'<text class="plan__note" x="{ax - 4:.1f}" y="{ay - 12:.1f}" style="text-anchor:start">{t["lane"]}</text>'
    else:
        arrow = f'<path d="M{ax:.1f} {ay - 6:.1f} L{ax:.1f} {ay + 16:.1f} M{ax - 6:.1f} {ay + 9:.1f} L{ax:.1f} {ay + 16:.1f} L{ax + 6:.1f} {ay + 9:.1f}" class="plan-arrow"/>'
        lane = f'<text class="plan__note" x="{ax + 14:.1f}" y="{ay + 4:.1f}" style="text-anchor:start">{t["lane"]}</text>'

    return (
        f'<svg class="plan__svg plan__svg--{orient}" viewBox="0 0 {vw:.0f} {vh:.0f}" role="group" aria-label="{t["plan_label"]}">'
        f'<g aria-hidden="true">{"".join(g_wash)}</g>'
        f'<g class="plan-ink" aria-hidden="true">{"".join(g_ink)}{arrow}</g>'
        f'<g aria-hidden="true">{lane}</g>'
        f'{"".join(g_hit)}</svg>'
    )
