# ---------------------------------------------------------------- drawing helpers
INK = "#3b2e24"


def jit(rng, a=0.9):
    return rng.uniform(-a, a)


def sketch_line(rng, x1, y1, x2, y2, w=1.3, extra=""):
    """Two slightly different strokes with overshoot: the hand-drawn ink line of the sketch."""
    out = []
    for k in range(2):
        dx, dy = x2 - x1, y2 - y1
        n = max((dx * dx + dy * dy) ** 0.5, 1)
        ox, oy = dx / n * 3, dy / n * 3  # overshoot at both ends
        mx, my = (x1 + x2) / 2 + jit(rng, 1.2), (y1 + y2) / 2 + jit(rng, 1.2)
        out.append(
            f'<path d="M{x1 - ox + jit(rng):.1f} {y1 - oy + jit(rng):.1f} Q{mx:.1f} {my:.1f} '
            f'{x2 + ox + jit(rng):.1f} {y2 + oy + jit(rng):.1f}" stroke-width="{w if k == 0 else w * 0.6:.2f}" '
            f'opacity="{0.95 if k == 0 else 0.55}"{extra}/>'
        )
    return "".join(out)


def sketch_rect(rng, x, y, w, h, sw=1.3):
    return (sketch_line(rng, x, y, x + w, y, sw) + sketch_line(rng, x + w, y, x + w, y + h, sw)
            + sketch_line(rng, x + w, y + h, x, y + h, sw) + sketch_line(rng, x, y + h, x, y, sw))


def tree(rng, cx, cy, r, cls="plan-tree"):
    spokes = "".join(
        f'<path d="M{cx:.1f} {cy:.1f} L{cx + r * 0.92 * c:.1f} {cy + r * 0.92 * s:.1f}"/>'
        for c, s in [(__import__("math").cos(a), __import__("math").sin(a))
                     for a in [k * 6.283 / 9 + rng.uniform(0, 0.4) for k in range(9)]]
    )
    return (f'<g class="{cls}"><circle cx="{cx:.1f}" cy="{cy:.1f}" r="{r:.1f}" class="plan-tree__wash" filter="url(#plan-wash)"/>'
            f'<circle cx="{cx:.1f}" cy="{cy:.1f}" r="{r:.1f}" class="plan-tree__ring"/>'
            f'<g class="plan-tree__spokes">{spokes}</g></g>')


def plan_svg(lang, orient):
    """The plan, horizontal (front at left) or vertical (front at top)."""
    t = T[lang]
    rng = random.Random(7)
    pad = 26
    if orient == "h":
        s, su = 4.4, 4.6
        def P(u, v):
            return pad + 34 + u * su, pad + v * s
        def R(u0, u1, v0, v1):
            x0, y0 = P(u0, v0); x1, y1 = P(u1, v1)
            return x0, y0, x1 - x0, y1 - y0
        vw, vh = pad * 2 + 34 + L * su, pad * 2 + W * s + 14
    else:
        s, su = 5.7, 4.15
        def P(u, v):
            return pad + v * s, pad + 34 + u * su
        def R(u0, u1, v0, v1):
            x0, y0 = P(u0, v0); x1, y1 = P(u1, v1)
            return x0, y0, x1 - x0, y1 - y0
        vw, vh = pad * 2 + W * s, pad * 2 + 34 + L * su + 14

    g_wash, g_ink, g_hit, g_text = [], [], [], []
    # trees around the plot (outside), like the sketch
    for (u, v, r) in [(-4, -6, 15), (30, 68, 13), (100, -8, 12), (172, 68, 16), (186, 12, 14), (128, 68, 11)]:
        x, y = P(u, v)
        g_wash.append(tree(rng, x, y, r * (1 if orient == "h" else 1.1), "plan-tree plan-tree--out"))
    def inside(pts, x, y):
        hit = False
        for k in range(len(pts)):
            (x1, y1), (x2, y2) = pts[k], pts[k - 1]
            if (y1 > y) != (y2 > y) and x < x1 + (y - y1) * (x2 - x1) / (y2 - y1):
                hit = not hit
        return hit

    def inset(pts, d):
        mx = sum(p[0] for p in pts) / len(pts); my = sum(p[1] for p in pts) / len(pts)
        out = []
        for (px, py) in pts:
            dx, dy = mx - px, my - py
            n = max((dx * dx + dy * dy) ** 0.5, 1)
            out.append((px + dx / n * d * 1.4, py + dy / n * d * 1.4))
        return out

    def pstr(pts):
        return " ".join(f"{px:.1f},{py:.1f}" for px, py in pts)

    for key, kind, shape, photo, en, vi, _, _ in ROOMS:
        if isinstance(shape, tuple):
            u0, u1, v0, v1 = shape
            shape = [(u0, v0), (u1, v0), (u1, v1), (u0, v1)]
        pts = [P(u, v) for u, v in shape]
        xs = [p[0] for p in pts]; ys = [p[1] for p in pts]
        x, y, w, h = min(xs), min(ys), max(xs) - min(xs), max(ys) - min(ys)
        outline = "".join(sketch_line(rng, *pts[k], *pts[(k + 1) % len(pts)], 1.1 if kind == "garden" else 2.0) for k in range(len(pts)))
        name = en if lang == "en" else vi
        if kind == "garden":
            g_wash.append(f'<polygon points="{pstr(inset(pts, 3))}" class="plan-wash plan-wash--garden" filter="url(#plan-wash)"/>')
            # planting in the garden, kept clear of the label in the middle
            n_tree = 4 if key == "front-garden" else 2
            placed, tries = 0, 0
            mx = sum(xs) / len(xs); my = sum(ys) / len(ys)
            while placed < n_tree and tries < 200:
                tries += 1
                r = rng.uniform(8, 12)
                tx = x + w * rng.uniform(0.1, 0.9); ty = y + h * rng.uniform(0.15, 0.85)
                if not all(inside(pts, tx + ddx, ty + ddy) for ddx, ddy in [(r, 0), (-r, 0), (0, r), (0, -r)]):
                    continue
                if abs(tx - mx) < 50 and abs(ty - my) < 22:
                    continue
                g_wash.append(tree(rng, tx, ty, r)); placed += 1
            g_ink.append(f'<g class="plan-wall plan-wall--fence">{outline}</g>')
        else:
            cls = {"room": "plan-wash--room", "wet": "plan-wash--wet"}[kind]
            g_wash.append(f'<polygon points="{pstr(inset(pts, 2))}" class="plan-wash {cls}" filter="url(#plan-wash)"/>'
                          f'<polygon points="{pstr(inset(pts, 4))}" class="plan-edge {cls}" filter="url(#plan-wash)"/>')
            if kind == "wet":
                g_wash.append(f'<polygon points="{pstr(pts)}" fill="url(#plan-hatch)"/>')
            g_ink.append(f'<g class="plan-wall">{outline}</g>')
        # the tap target and its label
        g_hit.append(
            f'<a class="plan__room" href="#room-{key}" data-room="{key}" aria-label="{name}">'
            f'<polygon class="plan__hit" points="{pstr(pts)}"/>'
        )
        cx, cy = sum(xs) / len(xs), sum(ys) / len(ys)
        if len(pts) == 3:
            w = w * 0.55
        if False:
            label = f'<text class="plan__label" x="{cx:.1f}" y="{cy:.1f}" transform="rotate(-90 {cx:.1f} {cy:.1f})">{name}</text>'
        else:
            words = name.split(" ")
            lines, cur = [], ""
            maxc = max(int(w / 7.2), 6)
            for wd in words:
                if len(cur) + len(wd) + 1 > maxc and cur:
                    lines.append(cur); cur = wd
                else:
                    cur = (cur + " " + wd).strip()
            lines.append(cur)
            y0 = cy - (len(lines) - 1) * 8
            label = "".join(f'<text class="plan__label" x="{cx:.1f}" y="{y0 + n * 16:.1f}">{ln}</text>' for n, ln in enumerate(lines))
        g_hit.append(label + "</a>")

    # details: the corridor, kitchen counter and table, beds, bathroom fixtures, drying lines
    hx, hy, hw, hh = R(90, 112, 25, 35)
    g_ink.append(f'<g class="plan-wall">{sketch_line(rng, *P(90, 25), *P(112, 25), 1.2)}{sketch_line(rng, *P(90, 35), *P(112, 35), 1.2)}</g>')
    hcx, hcy = hx + hw / 2, hy + hh / 2 + 3
    g_ink.append(f'<text class="plan__note" x="{hcx:.1f}" y="{hcy:.1f}">{t["hall"]}</text>')
    kx, ky, kw, kh = R(6, 70, 55, 59)  # kitchen counter along the outer wall
    g_ink.append(f'<rect x="{kx:.1f}" y="{ky:.1f}" width="{kw:.1f}" height="{kh:.1f}" class="plan-furn"/>')
    tx, ty = P(78, 46)
    g_ink.append(f'<circle cx="{tx:.1f}" cy="{ty:.1f}" r="12" class="plan-furn"/>')
    for (u0, u1, v0, v1) in [(138, 160, 4, 24), (120, 142, 36, 54)]:
        bx, by, bw, bh = R(u0, u1, v0, v1)
        g_ink.append(f'<rect x="{bx:.1f}" y="{by:.1f}" width="{bw:.1f}" height="{bh:.1f}" rx="2" class="plan-furn"/>')
    wx, wy = P(96, 6)
    g_ink.append(f'<circle cx="{wx:.1f}" cy="{wy:.1f}" r="5" class="plan-furn"/>')
    for k in range(3):
        lx0, ly0 = P(93 + k * 7, 40); lx1, ly1 = P(93 + k * 7, 56)
        g_ink.append(f'<path d="M{lx0:.1f} {ly0:.1f} L{lx1:.1f} {ly1:.1f}" class="plan-sky"/>')
    # the lane and the arrow in, like the sketch
    ax, ay = P(-12, 17)
    if orient == "h":
        arrow = f'<path d="M{ax - 6:.1f} {ay:.1f} L{ax + 16:.1f} {ay:.1f} M{ax + 9:.1f} {ay - 6:.1f} L{ax + 16:.1f} {ay:.1f} L{ax + 9:.1f} {ay + 6:.1f}" class="plan-arrow"/>'
        lane = f'<text class="plan__note" x="{ax - 4:.1f}" y="{ay - 12:.1f}" style="text-anchor:start">{t["lane"]}</text>'
    else:
        arrow = f'<path d="M{ax:.1f} {ay - 6:.1f} L{ax:.1f} {ay + 16:.1f} M{ax - 6:.1f} {ay + 9:.1f} L{ax:.1f} {ay + 16:.1f} L{ax + 6:.1f} {ay + 9:.1f}" class="plan-arrow"/>'
        lane = f'<text class="plan__note" x="{ax + 14:.1f}" y="{ay + 4:.1f}" style="text-anchor:start">{t["lane"]}</text>'

    return (
        f'<svg class="plan__svg plan__svg--{orient}" viewBox="0 0 {vw:.0f} {vh:.0f}" role="group" aria-label="{t["plan_label"]}">'
        f'<g aria-hidden="true">{"".join(g_wash)}</g>'
        f'<g class="plan-ink" aria-hidden="true">{"".join(g_ink)}{arrow}</g>'
        f'<g aria-hidden="true">{lane}</g>'
        f'{"".join(g_hit)}</svg>'
    )
